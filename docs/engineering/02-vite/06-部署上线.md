# 部署上线 + CI/CD 自动化

## 一、部署概述

::: tip 部署目标
企业级项目的部署目标是：**自动化、可回滚、高可用、可监控**。
:::

### 1、部署方式

::: info 部署方式对比

| 方式 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| **静态托管** | 简单、快速、便宜 | 功能受限 | 静态网站 |
| **服务器部署** | 灵活、可控 | 需要维护 | 企业应用 |
| **容器化部署** | 隔离、可移植 | 配置复杂 | 微服务 |
| **Serverless** | 按需付费、自动扩缩容 | 冷启动 | 轻量应用 |

:::

### 2、CI/CD 流程

::: details CI/CD 流程

```
代码提交 → 触发 CI → 构建测试 → 构建镜像 → 部署到测试环境 
→ 测试通过 → 部署到生产环境 → 监控告警
```

:::

## 二、静态资源部署

### 1、Nginx 部署

#### 1.1、Nginx 配置

```nginx
# /etc/nginx/conf.d/vite-app.conf
server {
    listen 80;
    server_name your-domain.com;
    root /var/www/vite-app/dist;
    index index.html;

    # Gzip 压缩
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css text/xml text/javascript application/javascript application/xml+rss application/json;

    # 静态资源缓存
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # HTML 不缓存
    location ~* \.(html)$ {
        expires -1;
        add_header Cache-Control "no-cache, no-store, must-revalidate";
    }

    # SPA 路由支持
    location / {
        try_files $uri $uri/ /index.html;
    }

    # API 代理
    location /api {
        proxy_pass http://backend-server:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
```

#### 1.2、部署脚本

```bash
#!/bin/bash
# deploy.sh

# 1. 构建项目
echo "开始构建..."
npm run build

# 2. 备份旧版本
if [ -d "/var/www/vite-app/dist" ]; then
    echo "备份旧版本..."
    tar -czf /var/www/backup/vite-app-$(date +%Y%m%d-%H%M%S).tar.gz /var/www/vite-app/dist
fi

# 3. 部署新版本
echo "部署新版本..."
rm -rf /var/www/vite-app/dist
cp -r dist /var/www/vite-app/

# 4. 重启 Nginx
echo "重启 Nginx..."
sudo systemctl reload nginx

echo "部署完成！"
```

### 2、CDN 部署

#### 2.1、上传到 CDN

```typescript
// scripts/upload-cdn.ts
import { S3Client, PutObjectCommand } from '@aws-sdk/client-s3';
import { readdirSync, readFileSync, statSync } from 'fs';
import { join } from 'path';

const s3 = new S3Client({
  region: 'us-east-1',
  credentials: {
    accessKeyId: process.env.AWS_ACCESS_KEY_ID!,
    secretAccessKey: process.env.AWS_SECRET_ACCESS_KEY!,
  },
});

async function uploadToCDN(dir: string, bucket: string) {
  const files = getAllFiles(dir);
  
  for (const file of files) {
    const content = readFileSync(file);
    const key = file.replace(dir + '/', '');
    
    await s3.send(
      new PutObjectCommand({
        Bucket: bucket,
        Key: key,
        Body: content,
        ContentType: getContentType(file),
        CacheControl: getCacheControl(file),
      })
    );
    
    console.log(`上传: ${key}`);
  }
}

function getAllFiles(dir: string): string[] {
  const files: string[] = [];
  const items = readdirSync(dir);
  
  for (const item of items) {
    const path = join(dir, item);
    const stat = statSync(path);
    
    if (stat.isDirectory()) {
      files.push(...getAllFiles(path));
    } else {
      files.push(path);
    }
  }
  
  return files;
}

function getContentType(file: string): string {
  const ext = file.split('.').pop();
  const types: Record<string, string> = {
    html: 'text/html',
    js: 'application/javascript',
    css: 'text/css',
    png: 'image/png',
    jpg: 'image/jpeg',
  };
  return types[ext || ''] || 'application/octet-stream';
}

function getCacheControl(file: string): string {
  if (file.endsWith('.html')) {
    return 'no-cache';
  }
  return 'public, max-age=31536000, immutable';
}

uploadToCDN('./dist', 'my-cdn-bucket');
```

## 三、Docker 容器化部署

### 1、Dockerfile

#### 1.1、多阶段构建

```dockerfile
# Dockerfile
# 阶段1: 构建
FROM node:18-alpine AS builder

WORKDIR /app

# 复制依赖文件
COPY package*.json ./
RUN npm ci

# 复制源代码
COPY . .

# 构建
RUN npm run build

# 阶段2: 运行
FROM nginx:alpine

# 复制构建产物
COPY --from=builder /app/dist /usr/share/nginx/html

# 复制 Nginx 配置
COPY nginx.conf /etc/nginx/conf.d/default.conf

EXPOSE 80

CMD ["nginx", "-g", "daemon off;"]
```

## 四、CI/CD 自动化

### 1、GitHub Actions

#### 1.1、基础配置

```yaml
# .github/workflows/deploy.yml
name: Deploy

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '18'
          cache: 'npm'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Run tests
        run: npm test
      
      - name: Build
        run: npm run build
      
      - name: Upload artifacts
        uses: actions/upload-artifact@v3
        with:
          name: dist
          path: dist
```

#### 1.2、部署到服务器

```yaml
# .github/workflows/deploy-production.yml
name: Deploy to Production

on:
  push:
    branches:
      - main

jobs:
  deploy:
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '18'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Build
        run: npm run build
        env:
          VITE_API_URL: ${{ secrets.API_URL }}
      
      - name: Deploy to server
        uses: appleboy/scp-action@master
        with:
          host: ${{ secrets.HOST }}
          username: ${{ secrets.USERNAME }}
          key: ${{ secrets.SSH_KEY }}
          source: "dist/*"
          target: "/var/www/vite-app"
      
      - name: Restart Nginx
        uses: appleboy/ssh-action@master
        with:
          host: ${{ secrets.HOST }}
          username: ${{ secrets.USERNAME }}
          key: ${{ secrets.SSH_KEY }}
          script: |
            sudo systemctl reload nginx
```

#### 1.3、Docker 部署

```yaml
# .github/workflows/docker-deploy.yml
name: Docker Deploy

on:
  push:
    branches:
      - main

          host: ${{ secrets.HOST }}
          username: ${{ secrets.USERNAME }}
```

```yaml
# .gitlab-ci.yml
stages:
  - build
  - test
  - deploy

variables:
  NODE_VERSION: "18"

  stage: build
  image: node:${NODE_VERSION}
  script:
    - npm ci

  image: node:${NODE_VERSION}

    - npm ci
    - npm test

  stage: deploy
  image: alpine:latest
  only:
    - develop
  script:
    - apk add --no-cache openssh-client
    - eval $(ssh-agent -s)
    - echo "$SSH_PRIVATE_KEY" | tr -d '\r' | ssh-add -
    - mkdir -p ~/.ssh
    - chmod 700 ~/.ssh
    - ssh-keyscan $STAGING_HOST >> ~/.ssh/known_hosts
    - scp -r dist/* $STAGING_USER@$STAGING_HOST:/var/www/staging
    - ssh $STAGING_USER@$STAGING_HOST "sudo systemctl reload nginx"
  environment:
    name: staging
    url: https://staging.example.com

  stage: deploy
  image: alpine:latest
  only:
    - main

    - apk add --no-cache openssh-client
```

### 3、Jenkins Pipeline

#### 3.1、Jenkinsfile

```groovy
// Jenkinsfile
pipeline {
    agent any
    
    environment {
        NODE_VERSION = '18'
        DOCKER_IMAGE = 'vite-app'
        DOCKER_REGISTRY = 'registry.example.com'
    }
    
    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }
        
        stage('Install') {
            steps {
                sh 'npm ci'
            }
        }
        
        stage('Test') {
            steps {
                sh 'npm test'
            }
        }
        
        stage('Build') {
            steps {
                sh 'npm run build'
            }
        }
        
        stage('Docker Build') {
            steps {
                script {
                    def image = docker.build("${DOCKER_REGISTRY}/${DOCKER_IMAGE}:${env.BUILD_NUMBER}")
                    image.push()
                    image.push("latest")
                }
            }
        }
        
        stage('Deploy') {
            steps {
                sh '''
                    docker stop vite-app || true
                    docker rm vite-app || true
                    docker run -d -p 80:80 --name vite-app ${DOCKER_REGISTRY}/${DOCKER_IMAGE}:${env.BUILD_NUMBER}
                '''
            }
        }
    }
    
    post {
        success {
            echo '部署成功！'
        }
        failure {
            echo '部署失败！'
        }
    }
}
```

### 2、日志收集

```typescript
// src/utils/logger.ts
export class Logger {
  private static sendLog(level: string, message: string, data?: any) {
    fetch('/api/logs', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        level,
        message,
        data,
        timestamp: new Date().toISOString(),
        url: window.location.href,
      }),
    });
  }
  
  static info(message: string, data?: any) {
    console.info(message, data);
    this.sendLog('info', message, data);
  }
  
  static error(message: string, error?: Error) {
    console.error(message, error);
    this.sendLog('error', message, { error: error?.stack });
  }
}
```

## 七、回滚策略

### 1、版本管理

```bash
#!/bin/bash
# rollback.sh

VERSION=$1

if [ -z "$VERSION" ]; then
    echo "用法: ./rollback.sh <version>"
    exit 1
fi

# 恢复指定版本
tar -xzf /var/www/backup/vite-app-${VERSION}.tar.gz -C /var/www/vite-app/

# 重启服务
sudo systemctl reload nginx

echo "已回滚到版本: ${VERSION}"
```

### 2、Docker 回滚

```bash
# 回滚到之前的镜像
docker stop vite-app
docker rm vite-app
docker run -d -p 80:80 --name vite-app vite-app:previous-version
```
