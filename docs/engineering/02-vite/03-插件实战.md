# 全体系插件实战 + 自定义插件开发

## 一、Vite 插件系统概述

::: tip 插件系统
Vite 插件系统基于 Rollup 插件接口，提供了丰富的钩子函数，允许开发者在构建和开发过程的各个阶段进行自定义处理。
:::

### 1、插件的作用

::: info 插件能力
- 🔧 **代码转换**：转换 TypeScript、JSX、Vue 等
- 📦 **依赖处理**：处理外部依赖、CDN 引入
- 🎨 **样式处理**：CSS 预处理器、PostCSS
- 🔍 **代码分析**：ESLint、TypeScript 检查
- 📊 **性能优化**：代码压缩、Tree Shaking
- 🛠️ **开发工具**：HMR、错误提示
:::

### 2、插件类型

::: details 插件分类

| 类型 | 说明 | 示例 |
|------|------|------|
| **官方插件** | Vite 官方维护 | @vitejs/plugin-react |
| **社区插件** | 社区贡献 | vite-plugin-pwa |
| **自定义插件** | 项目特定需求 | 内部业务插件 |

:::

### 3、实战案例

#### 3.1、自动导入插件

```typescript
// plugins/auto-import.ts
import type { Plugin } from 'vite';
import { readdirSync, statSync } from 'fs';
import { join, extname } from 'path';

interface AutoImportOptions {
  dirs: string[];
  prefix?: string;
}

export function autoImport(options: AutoImportOptions): Plugin {
  const { dirs, prefix = '' } = options;
  
  return {
    name: 'auto-import',
    resolveId(id) {
      if (id.startsWith('auto-import:')) {
        return id;
      }
    },
    load(id) {
      if (id.startsWith('auto-import:')) {
        const dir = id.replace('auto-import:', '');
        const files = getFiles(dir);
        
        const imports = files
          .map((file, index) => {
            const name = getComponentName(file);
            return `import ${name} from '${file}';`;
          })
          .join('\n');
        
        const exports = files
          .map((file) => getComponentName(file))
          .join(',\n  ');
        
        return `
${imports}

export {
  ${exports}
};
        `;
      }
    },
  };
  
  function getFiles(dir: string): string[] {
    const files: string[] = [];
    const fullPath = join(process.cwd(), dir);
    
    function traverse(currentPath: string) {
      const items = readdirSync(currentPath);
      
      for (const item of items) {
        const itemPath = join(currentPath, item);
        const stat = statSync(itemPath);
        
        if (stat.isDirectory()) {
          traverse(itemPath);
        } else if (stat.isFile() && extname(item) === '.tsx') {
          files.push(itemPath);
        }
      }
    }
    
    traverse(fullPath);
    return files;
  }
  
  function getComponentName(filePath: string): string {
    const name = filePath.split('/').pop()?.replace('.tsx', '') || '';
    return prefix + name;
  }
}
```

#### 3.2、Mock 数据插件

```typescript
// plugins/mock-plugin.ts
import type { Plugin } from 'vite';
import { readFileSync } from 'fs';
import { join } from 'path';

interface MockOptions {
  mockDir?: string;
}

export function mockPlugin(options: MockOptions = {}): Plugin {
  const { mockDir = 'mock' } = options;
  
  return {
    name: 'mock-plugin',
    configureServer(server) {
      server.middlewares.use('/api', (req, res, next) => {
        const method = req.method?.toLowerCase();
        const path = req.url;
        
        // 查找对应的 mock 文件
        const mockFile = join(process.cwd(), mockDir, `${method}${path}.json`);
        
        try {
          const data = readFileSync(mockFile, 'utf-8');
          res.setHeader('Content-Type', 'application/json');
          res.end(data);
        } catch {
          next();
        }
      });
    },
  };
}
```

#### 3.3、图片压缩插件

```typescript
// plugins/image-compress.ts
import type { Plugin } from 'vite';
import sharp from 'sharp';
import { readFileSync, writeFileSync } from 'fs';

export function imageCompressPlugin(): Plugin {
  return {
    name: 'image-compress',
    async generateBundle(options, bundle) {
      for (const fileName in bundle) {
        const chunk = bundle[fileName];
        
        if (chunk.type === 'asset' && /\.(png|jpg|jpeg)$/.test(fileName)) {
          const asset = chunk as any;
          const buffer = Buffer.from(asset.source);
          
          // 压缩图片
          const compressed = await sharp(buffer)
            .resize(1920, null, { withoutEnlargement: true })
            .jpeg({ quality: 80 })
            .toBuffer();
          
          asset.source = compressed;
        }
      }
    },
  };
}
```

#### 3.4、代码分析插件

```typescript
// plugins/analyze-plugin.ts
import type { Plugin } from 'vite';
import { build } from 'esbuild';

export function analyzePlugin(): Plugin {
  return {
    name: 'analyze-plugin',
    async buildEnd() {
      const result = await build({
        entryPoints: ['src/main.ts'],
        bundle: true,
        write: false,
        metafile: true,
      });
      
      // 分析依赖
      const analysis = {
        totalSize: 0,
        files: [] as Array<{ path: string; size: number }>,
      };
      
      if (result.metafile) {
        for (const [path, meta] of Object.entries(result.metafile.outputs)) {
          const size = meta.bytes;
          analysis.totalSize += size;
          analysis.files.push({ path, size });
        }
      }
      
      // 输出分析报告
      console.log('构建分析:');
      console.log(`总大小: ${(analysis.totalSize / 1024).toFixed(2)} KB`);
      analysis.files
        .sort((a, b) => b.size - a.size)
        .slice(0, 10)
        .forEach((file) => {
          console.log(`  ${file.path}: ${(file.size / 1024).toFixed(2)} KB`);
        });
    },
  };
}
```

## 四、插件开发最佳实践

### 1、插件设计原则

::: tip 设计原则
1. **单一职责**：每个插件只做一件事
2. **可配置**：提供灵活的配置选项
3. **性能优先**：避免不必要的处理
4. **错误处理**：完善的错误提示
5. **文档完善**：清晰的文档和示例
:::

### 2、类型安全

```typescript
// 定义插件选项类型
interface MyPluginOptions {
  enabled?: boolean;
  pattern?: RegExp;
  transform?: (code: string) => string;
}

export function myPlugin(options: MyPluginOptions = {}): Plugin {
  const {
    enabled = true,
    pattern = /\.ts$/,
    transform = (code) => code,
  } = options;
  
  return {
    name: 'my-plugin',
    transform(code, id) {
      if (!enabled || !pattern.test(id)) {
        return null;
      }
      
      return transform(code);
    },
  };
}
```

### 3、错误处理

```typescript
export function safePlugin(): Plugin {
  return {
    name: 'safe-plugin',
    transform(code, id) {
      try {
        // 转换逻辑
        return transformedCode;
      } catch (error) {
        this.error({
          message: `转换失败: ${error.message}`,
          id,
          plugin: 'safe-plugin',
        });
      }
    },
  };
}
```

### 4、调试技巧

```typescript
export function debugPlugin(): Plugin {
  return {
    name: 'debug-plugin',
    transform(code, id) {
      // 使用 this.warn 输出警告

    },
  };
}
```
