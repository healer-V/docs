# Vue 3 + React 18 双项目 Vite 配置实战

## 一、项目架构设计

### 1. 为什么用 Monorepo

企业中经常出现同时维护 Vue 和 React 项目的场景：

- 技术迁移期间，新功能用 React 开发，旧系统继续用 Vue 维护
- 不同产品线由不同团队负责，框架选型不同
- 需要将公共的类型定义、API 客户端、工具函数统一管理

Monorepo 让这两个项目**共享代码、统一构建流程**，同时保持各自的框架独立性。

### 2. 整体目录结构

```
vite-monorepo/
├── packages/
│   ├── shared/                # 共享代码（框架无关）
│   │   ├── src/
│   │   │   ├── api/           # API 客户端
│   │   │   ├── types/         # 类型定义
│   │   │   └── utils/         # 工具函数
│   │   └── package.json
│   ├── vue-admin/             # Vue 3 + Vite 管理后台
│   │   ├── src/
│   │   ├── vite.config.ts
│   │   └── package.json
│   └── react-portal/          # React 18 + Vite 用户门户
│       ├── src/
│       ├── vite.config.ts
│       └── package.json
├── package.json               # 根工作区配置
└── pnpm-workspace.yaml
```

## 二、Monorepo 基础搭建

### 1. pnpm workspace 配置

::: details 初始化步骤

```bash
# 1. 创建根目录
mkdir vite-monorepo && cd vite-monorepo

# 2. 初始化
pnpm init

# 3. 创建 workspace 配置
cat > pnpm-workspace.yaml << EOF
packages:
  - 'packages/*'
EOF

# 4. 创建各子包
mkdir -p packages/shared packages/vue-admin packages/react-portal
```

:::

根目录 `package.json`：

```json
{
  "name": "vite-monorepo",
  "private": true,
  "scripts": {
    "dev:vue": "pnpm --filter vue-admin dev",
    "dev:react": "pnpm --filter react-portal dev",
    "dev": "pnpm -r --parallel dev",
    "build:vue": "pnpm --filter vue-admin build",
    "build:react": "pnpm --filter react-portal build",
    "build": "pnpm -r --filter './packages/*' build"
  },
  "devDependencies": {
    "typescript": "^5.3.0"
  }
}
```

```ts
// 格式化日期
export function formatDate(date: Date | string, format = 'YYYY-MM-DD'): string {
  const d = new Date(date)
  const year = d.getFullYear()
  const month = String(d.getMonth() + 1).padStart(2, '0')
  const day = String(d.getDate()).padStart(2, '0')
  return format.replace('YYYY', String(year)).replace('MM', month).replace('DD', day)
}

// 防抖
export function debounce<T extends (...args: any[]) => any>(fn: T, delay: number) {
  let timer: ReturnType<typeof setTimeout>
  return (...args: Parameters<T>) => {
    clearTimeout(timer)
    timer = setTimeout(() => fn(...args), delay)
  }
}

// 深拷贝（简单场景使用）
export function deepClone<T>(obj: T): T {
  return JSON.parse(JSON.stringify(obj))
}
```

### 2. vite.config.ts

::: details vue-admin/vite.config.ts

```ts
// packages/vue-admin/vite.config.ts
import { defineConfig, loadEnv } from 'vite'
import vue from '@vitejs/plugin-vue'
import path from 'path'

export default defineConfig(({ mode }) => {
  const env = loadEnv(mode, process.cwd(), '')

  return {
    plugins: [vue()],

    resolve: {
      alias: {
        '@': path.resolve(__dirname, './src'),
        // 引用 shared 包的路径（使用 workspace 协议时无需特殊配置）
      },
    },

    server: {
      port: 3000,
      proxy: {
        '/api': {
          target: env.VITE_API_BASE_URL || 'http://localhost:8080',
          changeOrigin: true,
          rewrite: (p) => p.replace(/^\/api/, ''),
        },
      },
    },

    build: {
      outDir: 'dist',
      rollupOptions: {
        output: {
          manualChunks: {
            'vue-vendor': ['vue', 'vue-router', 'pinia'],
          },
        },
      },
    },
  }
})
```

### 3. 核心功能实现

::: details Vue 3 用户管理页面

```vue
<!-- packages/vue-admin/src/pages/UserManage.vue -->
<template>
  <div class="user-manage">
    <h2>用户管理</h2>

    <div v-if="loading">加载中...</div>

    <table v-else>
      <thead>
        <tr>
          <th>姓名</th>
          <th>邮箱</th>
          <th>角色</th>
          <th>创建时间</th>
        </tr>
      </thead>
      <tbody>
        <tr v-for="user in users" :key="user.id">
          <td>{{ user.name }}</td>
          <td>{{ user.email }}</td>
          <td>{{ user.role }}</td>
          <td>{{ formatDate(user.createdAt) }}</td>
        </tr>
      </tbody>
    </table>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { formatDate } from '@mono/shared/utils'
import type { User } from '@mono/shared/types'

const users = ref<User[]>([])
const loading = ref(false)

async function fetchUsers() {
  loading.value = true
  try {
    const res = await fetch('/api/users')
    const json = await res.json()
    users.value = json.data
  } finally {
    loading.value = false
  }
}

onMounted(fetchUsers)
</script>
```

```ts
// packages/react-portal/vite.config.ts
import { defineConfig, loadEnv } from 'vite'
import react from '@vitejs/plugin-react'
import path from 'path'

export default defineConfig(({ mode }) => {

        output: {
          manualChunks: {
            'react-vendor': ['react', 'react-dom'],
            'router': ['react-router-dom'],
          },
```

### 3. 核心功能实现

::: details React 18 用户管理页面

```tsx
// packages/react-portal/src/pages/UserManage.tsx
import { useState, useEffect } from 'react'
import { formatDate } from '@mono/shared/utils'
import type { User } from '@mono/shared/types'

export function UserManage() {
  const [users, setUsers] = useState<User[]>([])
  const [loading, setLoading] = useState(false)

  useEffect(() => {
    async function fetchUsers() {
      setLoading(true)
      try {
        const res = await fetch('/api/users')
        const json = await res.json()
        setUsers(json.data)
      } finally {
        setLoading(false)
      }
    }
    fetchUsers()
  }, [])

  if (loading) return <div>加载中...</div>

  return (
    <div className="user-manage">
      <h2>用户管理</h2>
      <table>
        <thead>
          <tr>
            <th>姓名</th><th>邮箱</th><th>角色</th><th>创建时间</th>
          </tr>
        </thead>
        <tbody>
          {users.map((user) => (
            <tr key={user.id}>
              <td>{user.name}</td>
              <td>{user.email}</td>
              <td>{user.role}</td>
              <td>{formatDate(user.createdAt)}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}
```

## 五、Vue 3 vs React 18 配置对比

### 1. Vite 配置差异

| 配置项 | Vue 3 | React 18 |
|--------|-------|----------|
| 核心插件 | `@vitejs/plugin-vue` | `@vitejs/plugin-react` 或 `plugin-react-swc` |
| JSX 支持 | 需要 `@vitejs/plugin-vue-jsx` | 内置（通过 Babel/SWC） |
| HMR 实现 | Vue HMR（基于组件实例） | Fast Refresh（基于 Hooks） |
| TS 检查命令 | `vue-tsc && vite build` | `tsc && vite build` |

### 2. 状态管理对比

::: code-group

```ts [Pinia（Vue 3）]
// packages/vue-admin/src/stores/userStore.ts
import { defineStore } from 'pinia'
import type { User } from '@mono/shared/types'

export const useUserStore = defineStore('user', {
  state: () => ({
    currentUser: null as User | null,
    isLoggedIn: false,
  }),
  actions: {
    async login(email: string, password: string) {
      const res = await fetch('/api/auth/login', {
        method: 'POST',
        body: JSON.stringify({ email, password }),
      })
      const { data } = await res.json()
      this.currentUser = data.user
      this.isLoggedIn = true
    },
    logout() {
      this.currentUser = null
      this.isLoggedIn = false
    },
  },
})
```

```ts [Zustand（React 18）]
// packages/react-portal/src/stores/userStore.ts
import { create } from 'zustand'
import type { User } from '@mono/shared/types'

interface UserStore {
  currentUser: User | null
  isLoggedIn: boolean
  login: (email: string, password: string) => Promise<void>
  logout: () => void
}

export const useUserStore = create<UserStore>((set) => ({
  currentUser: null,
  isLoggedIn: false,

  async login(email, password) {
    const res = await fetch('/api/auth/login', {
      method: 'POST',
      body: JSON.stringify({ email, password }),
    })
    const { data } = await res.json()
    set({ currentUser: data.user, isLoggedIn: true })
  },

  logout() {
    set({ currentUser: null, isLoggedIn: false })
  },
}))
```

## 六、最佳实践

### 1. 共享代码边界

::: tip 什么适合放在 shared 包
- 类型定义（TypeScript interfaces/types）
- 框架无关的工具函数（日期格式化、数字处理、本地存储封装）
- API 客户端基类（基于 fetch/axios 的请求封装）
- 常量定义（错误码、状态枚举）

不适合放在 shared：任何依赖框架 API 的代码（Vue 的 `ref`、React 的 `useState` 等）
:::

### 2. TypeScript 路径配置

每个子包都需要单独配置 `tsconfig.json` 以正确解析 `@mono/shared` 的类型：

```json
{
  "compilerOptions": {
    "baseUrl": ".",
    "paths": {
      "@/*": ["./src/*"],
      "@mono/shared": ["../shared/src/index.ts"],
      "@mono/shared/*": ["../shared/src/*"]
    }
  }
}
```

### 3. 统一版本管理

在根目录的 `package.json` 中统一管理共享的开发依赖版本，避免两个子项目使用不同版本的 TypeScript 或 ESLint：

```json
{
  "devDependencies": {
    "typescript": "^5.3.0",
    "eslint": "^8.57.0",
    "prettier": "^3.2.0"
  }
}
```
