```ts
  return {

    resolve: {
      alias: {
        // 步骤3：配置路径别名，@ 指向 src 目录
```

## 四、移动端屏幕适配

移动端适配方案选择 **`postcss-px-to-viewport`**：设计稿使用 px，构建时自动转换为 vw/vh，实现等比缩放。

### 4.1 PostCSS 配置

> 📄 **文件：`postcss.config.js`**（在项目根目录创建）

```js
export default {
  plugins: {
    // 步骤1：配置 postcss-px-to-viewport，将 px 自动转换为 vw
    'postcss-px-to-viewport-8-plugin': {
      // 设计稿宽度（UI 给的是多少就写多少，常见 375 或 750）
      viewportWidth: 375,

      // 不需要转换的 CSS 属性
      exclude: [/node_modules\/vant/],  // Vant 内部样式不转换（Vant 已适配移动端）

      // 单位精度（保留小数位数）
      unitPrecision: 5,

      // 需要转换的属性（* 表示全部）
      propList: ['*'],

      // 小于这个值的 px 不转换（通常 1px 边框不转换）
      minPixelValue: 1,

      // 媒体查询中的 px 是否也转换
      mediaQuery: false,
    },
  },
}
```

## 六、Axios 请求封装

> 📄 **文件：`src/utils/request.ts`**

```ts
import axios, { type AxiosRequestConfig, type AxiosResponse, type InternalAxiosRequestConfig } from 'axios'
import { showToast, showLoadingToast, closeToast } from 'vant'
import router from '@/router'
import { useUserStore } from '@/stores/modules/userStore'

// 步骤1：定义统一的接口响应数据结构
export interface ApiResponse<T = unknown> {
  code: number
  message: string
  data: T
}

// 步骤2：创建 Axios 实例，统一配置 baseURL 和超时时间
const request = axios.create({
  baseURL: import.meta.env.VITE_API_PREFIX,  // 统一前缀，走 Vite proxy
  timeout: 10000,  // 10 秒超时
  headers: {
    'Content-Type': 'application/json',
  },
})

// 步骤3：请求拦截器 — 在每个请求头中自动附带 Token
request.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    const userStore = useUserStore()

    // 如果有 token，自动写入请求头（不需要每个接口手动传）
    if (userStore.token) {
      config.headers.Authorization = `Bearer ${userStore.token}`
    }

    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// 步骤4：响应拦截器 — 统一处理业务错误码和 HTTP 错误
request.interceptors.response.use(
  (response: AxiosResponse<ApiResponse>) => {
    const { code, message, data } = response.data

    // 业务成功（根据实际后端约定修改状态码）
    if (code === 200) {
      return data as any
    }

    // 登录过期
    if (code === 401) {
      const userStore = useUserStore()
      userStore.logout()
      router.push('/login')
      showToast('登录已过期，请重新登录')
      return Promise.reject(new Error(message))
    }

    // 其他业务错误
    showToast(message || '请求失败')
    return Promise.reject(new Error(message))
  },
  (error) => {
    // HTTP 层错误（网络断开、超时等）
    const messageMap: Record<number, string> = {
      400: '请求参数错误',
      401: '未授权，请重新登录',
      403: '拒绝访问',
      404: '请求的资源不存在',
      500: '服务器内部错误',
      502: '网关错误',
      503: '服务不可用',
    }

    const status = error.response?.status
    const msg = messageMap[status] ?? '网络连接异常，请检查网络'
    showToast(msg)

    return Promise.reject(error)
  }
)

// 步骤5：封装 GET / POST / PUT / DELETE 方法，统一泛型类型
export const get = <T>(url: string, params?: object, config?: AxiosRequestConfig): Promise<T> => {
  return request.get(url, { params, ...config })
}

export const post = <T>(url: string, data?: object, config?: AxiosRequestConfig): Promise<T> => {
  return request.post(url, data, config)
}

export const put = <T>(url: string, data?: object, config?: AxiosRequestConfig): Promise<T> => {
  return request.put(url, data, config)
}

export const del = <T>(url: string, params?: object, config?: AxiosRequestConfig): Promise<T> => {
  return request.delete(url, { params, ...config })
}

export default request
```

> 📄 **文件：`src/router/guards.ts`**

```ts
import type { Router } from 'vue-router'
import { useUserStore } from '@/stores/modules/userStore'

// 步骤：路由守卫统一在此配置，不要散落在各个页面
export function setupRouterGuards(router: Router) {

  // 前置守卫
  router.beforeEach((to, from, next) => {
    const userStore = useUserStore()

    // 动态设置页面标题
    if (to.meta.title) {
      document.title = `${to.meta.title} - ${import.meta.env.VITE_APP_TITLE}`
    }

    // 需要登录但未登录，跳转登录页
    if (to.meta.requireAuth && !userStore.isLoggedIn) {
      next({ path: '/login', query: { redirect: to.fullPath } })
      return
    }

    // 已登录不允许再访问登录页
    if (to.path === '/login' && userStore.isLoggedIn) {
      next({ path: '/' })
      return
    }

    next()
  })

  // 后置守卫（页面切换完成后）
  router.afterEach(() => {
    // 滚动到顶部
    window.scrollTo(0, 0)
  })
}
```

## 九、Pinia 状态管理

> 📄 **文件：`src/stores/modules/userStore.ts`**

```ts
import { defineStore } from 'pinia'
import { getUserInfoApi, logoutApi } from '@/api'
import { storage } from '@/utils/storage'

interface UserInfo {
  id: number
  nickname: string
  avatar: string
  phone: string
}

interface UserState {
  token: string
  userInfo: UserInfo | null
}

export const useUserStore = defineStore('user', {
  state: (): UserState => ({
    token: storage.get('token') ?? '',  // 初始化时从 localStorage 恢复
    userInfo: null,
  }),

  getters: {
    // 步骤：通过 getters 派生状态，避免在组件中写重复的判断逻辑
    isLoggedIn: (state) => !!state.token,
    userName: (state) => state.userInfo?.nickname ?? '未登录',
  },

  actions: {
    // 保存登录信息
    setToken(token: string) {
      this.token = token
      storage.set('token', token)  // 持久化到 localStorage
    },

    // 获取用户信息
    async fetchUserInfo() {
      const userInfo = await getUserInfoApi()
      this.userInfo = userInfo
    },

    // 退出登录：清空所有用户状态
    async logout() {
      try {
        await logoutApi()
      } catch {
        // 即使接口失败也要清除本地状态
      } finally {
        this.token = ''
        this.userInfo = null
        storage.remove('token')
```

## 十、Storage 工具封装

> 📄 **文件：`src/utils/storage.ts`**

```ts
// 步骤：封装 localStorage，统一 key 前缀，防止 key 冲突，并加类型支持
const PREFIX = 'APP_'  // 修改为你的项目简称

const storage = {
  set<T>(key: string, value: T): void {
    const data = JSON.stringify({ value, timestamp: Date.now() })
    localStorage.setItem(PREFIX + key, data)
  },

  get<T>(key: string): T | null {
    const raw = localStorage.getItem(PREFIX + key)
    if (!raw) return null
    try {
      const { value } = JSON.parse(raw)
      return value as T
    } catch {
      return null
    }
  },

  remove(key: string): void {
    localStorage.removeItem(PREFIX + key)
  },

  clear(): void {
    // 只清除本项目的 key，不影响其他项目
    Object.keys(localStorage)
      .filter((key) => key.startsWith(PREFIX))
      .forEach((key) => localStorage.removeItem(key))
  },
}

export { storage }
```

## 十七、常用 Composable 封装

> 📄 **文件：`src/composables/useRequest.ts`**

```ts
import { ref } from 'vue'

// 步骤：封装通用的请求状态管理，避免每个页面重复写 loading / error
export function useRequest<T>(apiFn: (...args: any[]) => Promise<T>) {
  const data = ref<T | null>(null)
  const loading = ref(false)
  const error = ref<Error | null>(null)

  const execute = async (...args: any[]) => {
    loading.value = true
    error.value = null
    try {
      data.value = await apiFn(...args)
    } catch (err) {
      error.value = err as Error
    } finally {
      loading.value = false
    }
  }

  return { data, loading, error, execute }
}

// 使用示例：
// const { data: userInfo, loading, execute: fetchUser } = useRequest(getUserInfoApi)
// onMounted(() => fetchUser())
```

> 📄 **文件：`src/composables/useScroll.ts`**

```ts
// 步骤：封装上拉加载更多逻辑，配合 Vant List 组件使用
export function usePagination<T>(
  apiFn: (params: { page: number; pageSize: number }) => Promise<{ list: T[]; total: number }>
) {
  const list = ref<T[]>([])
  const page = ref(1)
  const pageSize = 20
  const loading = ref(false)
  const finished = ref(false)
  const total = ref(0)

  const loadMore = async () => {
    if (loading.value || finished.value) return

    loading.value = true
    try {
      const result = await apiFn({ page: page.value, pageSize })
      list.value = [...list.value, ...result.list]
      total.value = result.total
      page.value++

      if (list.value.length >= total.value) {
        finished.value = true
      }
    } finally {
      loading.value = false
    }
  }

  const refresh = async () => {
    list.value = []
    page.value = 1
    finished.value = false
    await loadMore()
  }

  return { list, loading, finished, total, loadMore, refresh }
}
```
